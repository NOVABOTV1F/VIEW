<script>
/* ---------- State ---------- */
let mediaRecorder=null, recordedChunks=[], stream=null, recording=false;
let recordStartTime=0, timerInterval=null, currentBlobUrl=null, currentBlob=null;
let history=JSON.parse(localStorage.getItem('xxyox_history')||'[]');
let pendingDeleteId=null;

/* ---------- Toast ---------- */
function toast(msg){
  const t=document.getElementById('toast');
  t.textContent=msg;
  t.classList.add('show');
  clearTimeout(t._h);
  t._h=setTimeout(()=>t.classList.remove('show'),2400);
}

/* ---------- Advanced settings toggle ---------- */
function toggleAdv(){
  document.getElementById('advBody').classList.toggle('open');
  document.getElementById('chev').classList.toggle('open');
}
document.getElementById('scrollSpeed').addEventListener('input',e=>{
  document.getElementById('scrollSpeedVal').textContent=e.target.value;
});

/* ---------- Recording ---------- */
async function toggleRecording(){
  if(recording){ stopRecording(); return; }
  const urlVal=document.getElementById('targetUrl').value.trim();
  const delaySec=parseInt(document.getElementById('delay').value)||0;

  try{
    stream=await navigator.mediaDevices.getDisplayMedia({
      video:{frameRate:parseInt(document.getElementById('fps').value)||30},
      audio:false
    });
  }catch(err){
    toast('Izin screen capture ditolak / dibatalkan');
    return;
  }

  if(delaySec>0){
    toast('Mulai dalam '+delaySec+' detik...');
    await new Promise(r=>setTimeout(r,delaySec*1000));
  }

  recordedChunks=[];
  let mime='video/webm;codecs=vp9';
  if(!MediaRecorder.isTypeSupported(mime)) mime='video/webm';
  mediaRecorder=new MediaRecorder(stream,{mimeType:mime});
  mediaRecorder.ondataavailable=e=>{ if(e.data.size>0) recordedChunks.push(e.data); };
  mediaRecorder.onstop=onRecordingStop;
  stream.getVideoTracks()[0].onended=()=>{ if(recording) stopRecording(); };

  mediaRecorder.start();
  recording=true;
  recordStartTime=Date.now();

  const preview=document.getElementById('preview');
  preview.srcObject=stream;
  preview.style.display='block';
  preview.play();
  document.getElementById('previewEmpty').style.display='none';
  document.getElementById('downloadBtn').style.display='none';

  const btn=document.getElementById('recordBtn');
  btn.classList.add('recording');
  document.getElementById('recLabel').textContent='Berhenti Rekam';
  document.getElementById('timer').style.display='block';
  timerInterval=setInterval(updateTimer,500);

  toast('Recording started');
}

function updateTimer(){
  const s=Math.floor((Date.now()-recordStartTime)/1000);
  const mm=String(Math.floor(s/60)).padStart(2,'0');
  const ss=String(s%60).padStart(2,'0');
  document.getElementById('timer').textContent=mm+':'+ss;
}

function stopRecording(){
  if(mediaRecorder && mediaRecorder.state!=='inactive') mediaRecorder.stop();
  if(stream) stream.getTracks().forEach(t=>t.stop());
  recording=false;
  clearInterval(timerInterval);
  document.getElementById('recordBtn').classList.remove('recording');
  document.getElementById('recLabel').textContent='Mulai Rekam';
}

function onRecordingStop(){
  const blob=new Blob(recordedChunks,{type:'video/webm'});
  currentBlob=blob;
  if(currentBlobUrl) URL.revokeObjectURL(currentBlobUrl);
  currentBlobUrl=URL.createObjectURL(blob);

  const preview=document.getElementById('preview');
  preview.srcObject=null;
  preview.src=currentBlobUrl;
  preview.muted=false;
  preview.controls=true;
  preview.play();

  document.getElementById('downloadBtn').style.display='flex';
  document.getElementById('timer').style.display='none';

  const name=document.getElementById('targetUrl').value.trim() || 'Rekaman Tanpa Judul';
  const settings=document.getElementById('resolution').value+' • '+
                  document.getElementById('fps').value+'fps';
  const entry={
    id:'r_'+Date.now(),
    name:name,
    date:new Date().toLocaleString('id-ID'),
    settings:settings,
    size:blob.size,
    url:currentBlobUrl,
    format:document.getElementById('exportFormat').value
  };
  history.unshift(entry);
  saveHistory();
  renderHistory();
  updateDashboard();
  toast('Recording selesai — siap diunduh');
}

function downloadCurrent(){
  if(!currentBlob) return;
  const a=document.createElement('a');
  a.href=currentBlobUrl;
  a.download='xxyox_recording_'+Date.now()+'.webm';
  a.click();
  toast('Download ready');
}

/* ---------- History ---------- */
function saveHistory(){
  const meta=history.map(h=>({id:h.id,name:h.name,date:h.date,settings:h.settings,size:h.size,format:h.format}));
  localStorage.setItem('xxyox_history',JSON.stringify(meta));
}

function fmtSize(bytes){
  if(bytes<1024) return bytes+' B';
  if(bytes<1024*1024) return (bytes/1024).toFixed(1)+' KB';
  return (bytes/1024/1024).toFixed(1)+' MB';
}

function renderHistory(){
  const list=document.getElementById('historyList');
  if(history.length===0){
    list.innerHTML='<div class="card"><div class="empty-state">Belum ada rekaman. Mulai rekam untuk melihat riwayat di sini.</div></div>';
    return;
  }
  list.innerHTML=history.map(h=>`
    <div class="hist-item" data-id="${h.id}">
      <div class="hist-thumb">
        <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="#B36A7A" stroke-width="2"><rect x="2" y="4" width="20" height="14" rx="2"/><path d="M8 21h8M12 18v3"/></svg>
      </div>
      <div class="hist-info">
        <div class="hist-name" id="name_${h.id}">${escapeHtml(h.name)}</div>
        <div class="hist-meta">${h.date} • ${h.settings} • ${fmtSize(h.size)}</div>
      </div>
      <button class="hist-menu-btn" onclick="toggleMenu('${h.id}')">⋮</button>
      <div class="hist-menu" id="menu_${h.id}">
        <button onclick="downloadHist('${h.id}')">⬇ Download</button>
        <button onclick="renameHist('${h.id}')">✎ Rename</button>
        <button class="danger" onclick="askDelete('${h.id}')">🗑 Delete</button>
      </div>
    </div>
  `).join('');
}

function escapeHtml(s){ const d=document.createElement('div'); d.textContent=s; return d.innerHTML; }

function toggleMenu(id){
  document.querySelectorAll('.hist-menu').forEach(m=>{
    if(m.id!=='menu_'+id) m.classList.remove('open');
  });
  document.getElementById('menu_'+id).classList.toggle('open');
}
document.addEventListener('click',e=>{
  if(!e.target.closest('.hist-menu-btn') && !e.target.closest('.hist-menu')){
    document.querySelectorAll('.hist-menu').forEach(m=>m.classList.remove('open'));
  }
});

function downloadHist(id){
  const h=history.find(x=>x.id===id);
  if(!h){ toast('File tidak tersedia lagi di sesi ini'); return; }
  if(h.url){
    const a=document.createElement('a');
    a.href=h.url; a.download=h.name.replace(/[^a-z0-9]/gi,'_')+'.webm'; a.click();
    toast('Download ready');
  }else{
    toast('Video ini tidak lagi tersimpan di memori (reload page)');
  }
}

function renameHist(id){
  const el=document.getElementById('name_'+id);
  const h=history.find(x=>x.id===id);
  el.innerHTML=`<input type="text" value="${escapeHtml(h.name)}" onblur="saveRename('${id}',this.value)" onkeydown="if(event.key==='Enter')this.blur()">`;
  el.querySelector('input').focus();
}
function saveRename(id,val){
  const h=history.find(x=>x.id===id);
  if(h && val.trim()){ h.name=val.trim(); saveHistory(); }
  renderHistory();
  toast('Settings saved');
}

function askDelete(id){ pendingDeleteId=id; document.getElementById('deleteModal').classList.add('open'); }
function closeDeleteModal(){ document.getElementById('deleteModal').classList.remove('open'); pendingDeleteId=null; }
function confirmDeleteOne(){
  history=history.filter(h=>h.id!==pendingDeleteId);
  saveHistory(); renderHistory(); updateDashboard();
  closeDeleteModal();
  toast('Video dihapus');
}

function openClearModal(){ document.getElementById('clearModal').classList.add('open'); }
function closeClearModal(){ document.getElementById('clearModal').classList.remove('open'); }
function confirmClearAll(){
  history=[]; saveHistory(); renderHistory(); updateDashboard();
  closeClearModal();
  toast('Semua riwayat dihapus');
}

/* ---------- Dashboard (simulated system metrics, real storage calc) ---------- */
function updateDashboard(){
  const totalBytes=history.reduce((a,h)=>a+(h.size||0),0);
  const localBytes=JSON.stringify(localStorage).length;
  const usedKB=(totalBytes+localBytes)/1024;
  const quotaKB=50*1024; // simulated 50MB quota display
  const pct=Math.min(100,(usedKB/quotaKB)*100);
  document.getElementById('storageTxt').textContent=fmtSize(totalBytes+localBytes);
  document.getElementById('storageBar').style.width=pct+'%';

  document.getElementById('totalVids').textContent=history.length;
  document.getElementById('totalSize').textContent=fmtSize(totalBytes);

  // RAM via performance.memory if available (Chrome), else simulated
  if(performance.memory){
    const used=performance.memory.usedJSHeapSize;
    const limit=performance.memory.jsHeapSizeLimit;
    const ramPct=Math.min(100,(used/limit)*100);
    document.getElementById('ramTxt').textContent=fmtSize(used);
    document.getElementById('ramBar').style.width=ramPct+'%';
  }else{
    const simRam=18+Math.random()*10;
    document.getElementById('ramTxt').textContent=simRam.toFixed(0)+'%';
    document.getElementById('ramBar').style.width=simRam+'%';
  }

  const simCpu=(recording?45:8)+Math.random()*12;
  document.getElementById('cpuTxt').textContent=simCpu.toFixed(0)+'%';
  document.getElementById('cpuBar').style.width=simCpu+'%';
}
setInterval(updateDashboard,3000);
